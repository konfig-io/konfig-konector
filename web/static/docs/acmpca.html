Example

# PrivateCA — ACM PCA
# Full-options example generated from the CRD schema. Optional fields are
# included with sample values; delete what you don't need. Field reference:
# https://konfig-konector.io/docs/acmpca.html#privateca
apiVersion: aws.konfig.io/v1alpha1
kind: PrivateCA
metadata:
  name: example-privateca
  namespace: default
spec:
  keyAlgorithm: RSA_2048
  permanentDeletionTimeInDays: 7
  providerRef:
    name: example-privateca
    region: us-east-1
  revocationConfiguration:
    crl:
      enabled: true
      expirationInDays: 1
      s3BucketName: example-privateca
  signingAlgorithm: SHA256WITHRSA
  subject:
    commonName: example-privateca
    country: example
    locality: example
    organization: example
    organizationalUnit: example
    state: example
  tags:
    environment: dev
    team: platform
  type: ROOT
  usageMode: GENERAL_PURPOSE
