Example

# CodePipeline — CodePipeline
# Full-options example generated from the CRD schema. Optional fields are
# included with sample values; delete what you don't need. Field reference:
# https://konfig-konector.io/docs/codepipeline.html#codepipeline
apiVersion: aws.konfig.io/v1alpha1
kind: CodePipeline
metadata:
  name: example-codepipeline
  namespace: default
spec:
  artifactStore:
    location: example
    type: S3
  pipelineName: example-codepipeline
  roleARN: arn:aws:iam::123456789012:role/example-role
  stages:
  - actions:
    - actionTypeID:
        category: example
        owner: example
        provider: example
        version: '1.0'
      inputArtifacts:
      - example
      name: example-codepipeline
      outputArtifacts:
      - example
      runOrder: 1
    name: example-codepipeline
  tags:
    environment: dev
    team: platform
