Example

# SQSQueue — SQS
# Full-options example generated from the CRD schema. Optional fields are
# included with sample values; delete what you don't need. Field reference:
# https://konfig-konector.io/docs/sqs.html#sqsqueue
apiVersion: aws.konfig.io/v1alpha1
kind: SQSQueue
metadata:
  name: example-sqsqueue
  namespace: default
spec:
  delaySeconds: 1
  fifo: false
  kmsKeyId: arn:aws:kms:us-east-1:123456789012:key/1234abcd-12ab-34cd-56ef-1234567890ab
  messageRetentionPeriod: 60
  policy: '{"Version": "2012-10-17", "Statement": [{"Effect": "Allow", "Action": ["s3:GetObject"], "Resource":
    "*"}]}'
  queueName: example-sqsqueue
  receiveMessageWaitTime: 1
  redrivePolicy:
    deadLetterQueueRef: example
    maxReceiveCount: 1
  tags:
    environment: dev
    team: platform
  visibilityTimeout: 1
