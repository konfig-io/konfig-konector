# CloudHSMCluster — CloudHSM

# included with sample values; delete what you don't need. Field reference:

metadata:
  name: example-cloudhsmcluster
  namespace: default

  backupRetentionPolicy:
    type: DAYS
    value: exa
  hsmType: example
  mode: FIPS
  networkType: IPV4
  providerRef:
    name: example-cloudhsmcluster
    region: us-east-1
  subnetIds:
  - example
  tags:
  - key: example
    value: example
