Example

# AutoScalingGroup — Auto Scaling
# Full-options example generated from the CRD schema. Optional fields are
# included with sample values; delete what you don't need. Field reference:
# https://konfig-konector.io/docs/autoscaling.html#autoscalinggroup
apiVersion: aws.konfig.io/v1alpha1
kind: AutoScalingGroup
metadata:
  name: example-autoscalinggroup
  namespace: default
spec:
  autoScalingGroupName: example-autoscalinggroup
  desiredCapacity: 2
  launchTemplateRef:
    name: example-autoscalinggroup
    version: '1.0'
  maxSize: 1
  minSize: 1
  providerRef:
    name: example-autoscalinggroup
    region: us-east-1
  tags:
    environment: dev
    team: platform
  targetGroupArns:
  - example
  vpcZoneIdentifier:
  - name: example-vpczoneidentifier
