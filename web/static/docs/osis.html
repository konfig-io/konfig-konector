  name: example-osispipeline
  namespace: default
spec:
  bufferOptions:
    persistentBufferEnabled: true
  encryptionAtRestOptions:
    kmsKeyArn: arn:aws:iam::123456789012:role/example-role
  logPublishingOptions:
    cloudWatchLogDestination:
      logGroup: example
    isLoggingEnabled: true
  maxUnits: 1
  minUnits: 1
  pipelineConfigurationBody: example
  pipelineName: example-osispipeline
  pipelineRoleArn: arn:aws:iam::123456789012:role/example-role
  providerRef:
    name: example-osispipeline
    region: us-east-1
  tags:
  - key: example
    value: example
  vpcOptions:
    securityGroupIds:
    - example
    subnetIds:
    - example
    vpcAttachmentOptions:
      attachToVpc: false
      cidrBlock: 10.0.0.0/16
    vpcEndpointManagement: CUSTOMER
