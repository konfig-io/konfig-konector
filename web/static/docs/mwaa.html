Example

# MWAAEnvironment — MWAA
# Full-options example generated from the CRD schema. Optional fields are

metadata:
  name: example-mwaaenvironment
  namespace: default
spec:
  airflowVersion: '1.0'
  dagS3Path: example
  endpointManagement: CUSTOMER
  environmentClass: example
  executionRoleArn: arn:aws:iam::123456789012:role/example-role
  kmsKey: example
  loggingConfiguration:
    dagProcessingLogs:
      cloudWatchLogGroupArn: arn:aws:iam::123456789012:role/example-role
      enabled: true
      logLevel: CRITICAL
    schedulerLogs:
      cloudWatchLogGroupArn: arn:aws:iam::123456789012:role/example-role
      enabled: true
      logLevel: CRITICAL
    taskLogs:
      cloudWatchLogGroupArn: arn:aws:iam::123456789012:role/example-role
      enabled: true
      logLevel: CRITICAL
    webserverLogs:
      cloudWatchLogGroupArn: arn:aws:iam::123456789012:role/example-role
      enabled: true
      logLevel: CRITICAL
    workerLogs:
      cloudWatchLogGroupArn: arn:aws:iam::123456789012:role/example-role
      enabled: true
      logLevel: CRITICAL
  maxWebservers: 1
  maxWorkers: 1
  minWebservers: 1
  minWorkers: 1
  name: example-mwaaenvironment
  networkConfiguration:
    securityGroupIds:
    - example
    subnetIds:
    - example
  pluginsS3ObjectVersion: '1.0'
  pluginsS3Path: example
  providerRef:
    name: example-mwaaenvironment
    region: us-east-1
  requirementsS3ObjectVersion: '1.0'
  requirementsS3Path: example
  schedulers: 1
  sourceBucketArn: arn:aws:iam::123456789012:role/example-role
  startupScriptS3ObjectVersion: '1.0'
  startupScriptS3Path: example
  webserverAccessMode: PRIVATE_ONLY
  weeklyMaintenanceWindowStart: example
  workerReplacementStrategy: FORCED
