Example

# Trail — CloudTrail
# Full-options example generated from the CRD schema. Optional fields are
# included with sample values; delete what you don't need. Field reference:
# https://konfig-konector.io/docs/cloudtrail.html#trail
apiVersion: aws.konfig.io/v1alpha1
kind: Trail
metadata:
  name: example-trail
  namespace: default
spec:
  cloudWatchLogsLogGroupArn: arn:aws:iam::123456789012:role/example-role
  cloudWatchLogsRoleArn: arn:aws:iam::123456789012:role/example-role
  enableLogFileValidation: true
  enableLogging: true
  includeGlobalServiceEvents: false
  isMultiRegionTrail: false
  kmsKeyId: arn:aws:kms:us-east-1:123456789012:key/1234abcd-12ab-34cd-56ef-1234567890ab
  providerRef:
    name: example-trail
    region: us-east-1
  s3BucketName: example-trail
  s3KeyPrefix: example
  tags:
    environment: dev
    team: platform
  trailName: example-trail
