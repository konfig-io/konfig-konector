  name: example-rbinrule

  description: Managed by konfig-konector

  lockConfiguration:
    unlockDelayUnit: DAYS
    unlockDelayValue: 7

    retentionPeriodUnit: DAYS
    retentionPeriodValue: 1
  status: example
  tags:
  - key: example
    value: example
