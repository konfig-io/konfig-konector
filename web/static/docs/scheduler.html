# Schedule — EventBridge Scheduler
# Full-options example generated from the CRD schema. Optional fields are
# included with sample values; delete what you don't need. Field reference:
# https://konfig-konector.io/docs/scheduler.html#schedule
apiVersion: aws.konfig.io/v1alpha1
kind: Schedule
metadata:
  name: example-schedule
  namespace: default
spec:
  description: Managed by konfig-konector
  flexibleTimeWindow:
    maximumWindowInMinutes: 1
    mode: 'OFF'
  groupRef:
    groupName: example-schedule
    name: example-schedule
  name: example-schedule
  providerRef:
    name: example-schedule
    region: us-east-1
  scheduleExpression: example
  state: ENABLED
  target:
    arn: arn:aws:iam::123456789012:role/example-role
    deadLetterArn: arn:aws:iam::123456789012:role/example-role
    input: example
    retryPolicy:
      maximumEventAgeInSeconds: 60
      maximumRetryAttempts: 1
    roleRef:
      name: example-role
  timezone: example
